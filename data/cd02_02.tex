\section{Nhu cầu năng lượng -- nhiên liệu}

\subsection{Năng lượng có ích và hiệu suất sử dụng năng lượng}

Trong thực tế, không phải toàn bộ năng lượng do nhiên liệu giải phóng đều được sử dụng cho mục đích mong muốn. Một phần năng lượng bị thất thoát ra môi trường.

\begin{ghinho}{Hiệu suất sử dụng năng lượng}
Nếu hiệu suất sử dụng năng lượng là $H\%$ thì:
\[
Q_{\text{ích}}=Q_{\text{tỏa}}\frac{H}{100},
\qquad
Q_{\text{tỏa}}=\frac{Q_{\text{ích}}}{H/100}.
\]
Cần phân biệt:
\begin{itemize}
\item \textbf{hiệu suất phản ứng}: mức độ chuyển hóa chất đầu thành sản phẩm;
\item \textbf{hiệu suất sử dụng năng lượng}: phần năng lượng tỏa ra thực sự được dùng cho mục đích cần thiết.
\end{itemize}
Hai loại hiệu suất có thể cùng xuất hiện trong một bài toán nhưng không được dùng thay cho nhau.
\end{ghinho}

\begin{vidu}
Một bếp cần cung cấp \SI{18}{\mega\joule} nhiệt có ích. Hiệu suất sử dụng nhiệt của bếp là $60\%$. Tính nhiệt lượng nhiên liệu phải tỏa ra.
\end{vidu}
\begin{loigiai}
Điểm đầu là năng lượng có ích, điểm đích là năng lượng nhiên liệu phải tỏa. Ta đi ngược qua bước hiệu suất:
\[
Q_{\text{tỏa}}=\frac{18}{0.60}=\SI{30}{\mega\joule}.
\]
\textbf{Vậy,} nhiên liệu phải tỏa ra \SI{30}{\mega\joule}.
\end{loigiai}

\subsection{Nhiên liệu}

Đây là một trong những chuỗi tính phổ biến nhất:
\[
\boxed{Q_{\text{ích}}}
\longrightarrow
\boxed{Q_{\text{tỏa}}}
\longrightarrow
\boxed{n\text{ hoặc }m\text{ nhiên liệu}}.
\]

\begin{ghinho}{Hướng giải dạng tính lượng nhiên liệu}
\begin{enumerate}
\item Xác định năng lượng có ích mà hệ cần.
\item Dùng hiệu suất để tính năng lượng nhiên liệu phải tỏa.
\item Dùng dữ kiện kJ/mol, kJ/g hoặc MJ/kg để tính lượng nhiên liệu.
\item Đổi sang đơn vị đề hỏi và kết luận.
\end{enumerate}
\end{ghinho}

\begin{vidu}
Một lò cần \SI{504}{\mega\joule} nhiệt có ích. Lò đốt methane, biết $1$ mol \ce{CH4} cháy hoàn toàn tỏa \SI{890}{\kilo\joule}; hiệu suất sử dụng nhiệt của lò là $70\%$. Tính khối lượng methane cần dùng. Cho $M_{\ce{CH4}}=16$ g/mol.
\end{vidu}
\begin{loigiai}
\textbf{Bước 1.} Năng lượng có ích cần dùng là \SI{504}{\mega\joule}.

\textbf{Bước 2.} Nhiệt lượng methane phải tỏa:
\[
Q_{\text{tỏa}}=\frac{504}{0.70}=\SI{720}{\mega\joule}=\SI{720000}{\kilo\joule}.
\]

\textbf{Bước 3.} Số mol methane:
\[
n_{\ce{CH4}}=\frac{720000}{890}\approx\SI{808.99}{\mole}.
\]
Khối lượng methane:
\[
m=808.99\times16\approx\SI{12944}{\gram}\approx\SI{12.94}{\kilo\gram}.
\]

\textbf{Vậy,} cần khoảng \SI{12.94}{\kilo\gram} methane.
\end{loigiai}

\subsection{Trao đổi nhiệt vật lí}

Một số đề không cho trực tiếp năng lượng cần thiết mà yêu cầu học sinh tự tính từ sự tăng nhiệt độ. Nếu một vật có khối lượng $m$, nhiệt dung riêng $c$ và tăng nhiệt độ từ $T_1$ lên $T_2$ thì:
\[
Q_{\text{cần}}=mc\Delta T,
\qquad
\Delta T=T_2-T_1.
\]

\begin{ghinho}{Hướng giải bài đun nóng}
\begin{enumerate}
\item Tính $\Delta T=T_2-T_1$.
\item Tính nhiệt lượng vật cần nhận: $Q_{\text{cần}}=mc\Delta T$.
\item Nếu có hiệu suất, tính $Q_{\text{nhiên liệu}}=Q_{\text{cần}}/(H/100)$.
\item Từ nhiệt trị suy ra lượng nhiên liệu cần dùng.
\end{enumerate}
\end{ghinho}

\begin{vidu}
Cần đun \SI{20}{\kilo\gram} nước từ \SI{25}{\degreeCelsius} đến \SI{100}{\degreeCelsius}. Biết nhiệt dung riêng của nước là \SI{4.2}{\joule\per\gram\per\degreeCelsius}. Bếp dùng nhiên liệu có nhiệt trị \SI{40}{\kilo\joule\per\gram}, hiệu suất $50\%$. Tính khối lượng nhiên liệu cần dùng.
\end{vidu}
\begin{loigiai}
\textbf{Bước 1.} Độ tăng nhiệt độ:
\[
\Delta T=100-25=\SI{75}{\degreeCelsius}.
\]

\textbf{Bước 2.} Đổi \SI{20}{\kilo\gram}=\SI{20000}{\gram}. Nhiệt lượng nước cần nhận:
\[
Q_{\text{cần}}=20000\times4.2\times75=\SI{6300000}{\joule}=\SI{6300}{\kilo\joule}.
\]

\textbf{Bước 3.} Do hiệu suất là $50\%$:
\[
Q_{\text{nhiên liệu}}=\frac{6300}{0.50}=\SI{12600}{\kilo\joule}.
\]

\textbf{Bước 4.} Khối lượng nhiên liệu:
\[
m=\frac{12600}{40}=\SI{315}{\gram}.
\]

\textbf{Vậy,} cần \SI{315}{\gram} nhiên liệu.
\end{loigiai}

\subsection{Trao đổi nhiệt hóa học}

Một phản ứng thu nhiệt cũng có thể được xem như một ``mục tiêu cần năng lượng''. Khi đó con đường giải hoàn toàn giống bài đun nóng:
\[
\boxed{\text{lượng chất cần xử lí}}
\to
\boxed{Q_{\text{phản ứng cần}}}
\to
\boxed{Q_{\text{nhiên liệu}}}
\to
\boxed{\text{lượng nhiên liệu}}.
\]

\begin{vidu}
Để phân hủy hoàn toàn $1$ mol \ce{CaCO3} cần cung cấp \SI{178}{\kilo\joule}. Một lò nung xử lí \SI{500}{\kilo\gram} \ce{CaCO3} tinh khiết. Lò dùng nhiên liệu có nhiệt trị \SI{40}{\mega\joule\per\kilo\gram}; hiệu suất truyền nhiệt là $50\%$. Tính khối lượng nhiên liệu tối thiểu cần dùng. Cho $M_{\ce{CaCO3}}=100$ g/mol.
\end{vidu}
\begin{loigiai}
Số mol \ce{CaCO3}:
\[
n=\frac{500000}{100}=\SI{5000}{\mole}.
\]
Năng lượng phản ứng cần:
\[
Q_{\text{cần}}=5000\times178=\SI{890000}{\kilo\joule}=\SI{890}{\mega\joule}.
\]
Nhiệt lượng nhiên liệu phải tỏa:
\[
Q_{\text{nhiên liệu}}=\frac{890}{0.50}=\SI{1780}{\mega\joule}.
\]
Khối lượng nhiên liệu:
\[
m=\frac{1780}{40}=\SI{44.5}{\kilo\gram}.
\]
\textbf{Vậy,} cần tối thiểu \SI{44.5}{\kilo\gram} nhiên liệu.
\end{loigiai}
