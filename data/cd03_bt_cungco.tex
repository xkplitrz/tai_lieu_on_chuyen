\subsection{Củng cố}

\begin{tuluyen}
Cho các chất: \ce{HCl}, \ce{NaOH}, \ce{NaCl}, \ce{Ca(OH)2}, \ce{H2SO4}. Phân loại thành acid, base và muối.
\dapso{Acid: \ce{HCl}, \ce{H2SO4}; base: \ce{NaOH}, \ce{Ca(OH)2}; muối: \ce{NaCl}.}
\end{tuluyen}

\begin{tuluyen}
Cho ba dung dịch có pH bằng 2, 7 và 11. Xác định môi trường của từng dung dịch.
\dapso{pH 2: acid; pH 7: trung tính; pH 11: base.}
\end{tuluyen}

\begin{tuluyen}
Viết phương trình hóa học cho các phản ứng sau nếu có:
\begin{tasks}(2)
\task \ce{Mg + HCl};
\task \ce{CuO + HCl};
\task \ce{CO2 + Ca(OH)2};
\task \ce{CuSO4 + NaOH};
\task \ce{BaCl2 + Na2SO4};
\task \ce{Na2CO3 + HCl};
\task \ce{NaHCO3 + HCl};
\task \ce{NaHCO3 + NaOH};
\task \ce{Ca(HCO3)2 + Ca(OH)2};
\task nhiệt phân \ce{NaHCO3}.
\end{tasks}
\dapso{Viết và cân bằng các phương trình; ghi kết tủa hoặc khí nếu có.}
\end{tuluyen}

\begin{tuluyen}
Nêu hiện tượng khi cho dung dịch \ce{NaOH} vào dung dịch \ce{CuSO4}. Viết phương trình hóa học.
\dapso{Xuất hiện kết tủa xanh \ce{Cu(OH)2}; \ce{CuSO4 + 2NaOH -> Cu(OH)2 v + Na2SO4}.}
\end{tuluyen}

\begin{tuluyen}
Viết các phương trình hóa học thực hiện chuỗi phản ứng:
\[
\begin{aligned}
\ce{CaCO3} &\ce{-> CaO -> Ca(OH)2 -> CaCO3}\\
&\ce{-> Ca(HCO3)2 -> CaCO3}.
\end{aligned}
\]
\dapso{Gồm các phản ứng nhiệt phân, hydrat hóa, carbonate hóa, hòa tan bằng \ce{CO2}/nước và tạo lại \ce{CaCO3}.}
\end{tuluyen}

\begin{tuluyen}
Hoàn thành chuỗi phản ứng:
\[
\ce{Cu -> CuO -> CuCl2 -> Cu(OH)2 -> CuO -> CuSO4}.
\]
\dapso{Dùng lần lượt \ce{O2}, \ce{HCl}, \ce{NaOH}, nhiệt và \ce{H2SO4}.}
\end{tuluyen}

\begin{tuluyen}
Có ba dung dịch không nhãn: \ce{HCl}, \ce{Na2CO3}, \ce{NaCl}. Chỉ dùng quỳ tím và dung dịch \ce{HCl} đã nhận biết được, hãy trình bày cách nhận biết cả ba dung dịch.
\dapso{Quỳ tím nhận \ce{HCl}; dùng \ce{HCl} nhận \ce{Na2CO3} qua khí \ce{CO2}; mẫu còn lại là \ce{NaCl}.}
\end{tuluyen}

\begin{tuluyen}
Trộn \SI{100}{\milli\liter} dung dịch \ce{HCl} \SI{0.20}{\mole\per\liter} với \SI{150}{\milli\liter} dung dịch \ce{NaOH} \SI{0.10}{\mole\per\liter}. Tính số mol chất dư.
\dapso{\ce{HCl} dư \SI{0.005}{\mole}.}
\begin{loigiaiBT}
\textbf{Bước 1. Tính số mol.}
\[
n_{\ce{HCl}}=0.100\times0.20=\SI{0.020}{\mole},
\]
\[
n_{\ce{NaOH}}=0.150\times0.10=\SI{0.015}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c}
\ce{HCl} & + & \ce{NaOH} & \ce{->} & \ce{NaCl} & + & \ce{H2O}\\
$0{,}020$ & & $0{,}015$ & & & & \\
$0{,}015$ & & $0{,}015$ & & $0{,}015$ & & $0{,}015$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\[
n_{\ce{HCl}\,\text{dư}}=0.020-0.015=\SI{0.005}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}

\textbf{Vậy,} sau phản ứng còn dư \SI{0.005}{\mole} \ce{HCl}.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Cho \SI{5.00}{\gram} \ce{CaCO3} phản ứng hoàn toàn với \ce{HCl} dư. Tính thể tích \ce{CO2} ở $V_m=\SI{24.8}{\liter\per\mole}$.
\dapso{\SI{1.24}{\liter}.}
\begin{loigiaiBT}
\textbf{Bước 1. Tính số mol.}
\[
n_{\ce{CaCO3}}=\frac{5.00}{100}=\SI{0.0500}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c c c}
\ce{CaCO3} & + & \ce{2HCl} & \ce{->} & \ce{CaCl2} & + & \ce{CO2} & + & \ce{H2O}\\
$0{,}0500$ & & & & $0{,}0500$ & & $0{,}0500$ & & $0{,}0500$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\[
n_{\ce{CO2}}=\SI{0.0500}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
V_{\ce{CO2}}=0.0500\times24.8=\SI{1.24}{\liter}.
\]

\textbf{Vậy,} thể tích \ce{CO2} thu được là \SI{1.24}{\liter}.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Trộn \SI{100}{\milli\liter} dung dịch \ce{CuSO4} \SI{0.30}{\mole\per\liter} với \SI{200}{\milli\liter} dung dịch \ce{NaOH} \SI{0.20}{\mole\per\liter}. Tính khối lượng kết tủa \ce{Cu(OH)2}. Cho $M=98$ g/mol.
\dapso{\SI{1.96}{\gram}.}
\begin{loigiaiBT}
\textbf{Bước 1. Tính số mol.}
\[
n_{\ce{CuSO4}}=0.100\times0.30=\SI{0.030}{\mole},
\]
\[
n_{\ce{NaOH}}=0.200\times0.20=\SI{0.040}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c}
\ce{CuSO4} & + & \ce{2NaOH} & \ce{->} & \ce{Cu(OH)2} & + & \ce{Na2SO4}\\
$0{,}030$ & & $0{,}040$ & & & & \\
$0{,}020$ & & $0{,}040$ & & $0{,}020$ & & $0{,}020$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\ce{NaOH} phản ứng hết trước nên:
\[
n_{\ce{Cu(OH)2}}=\frac{0.040}{2}=\SI{0.020}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
m_{\ce{Cu(OH)2}}=0.020\times98=\SI{1.96}{\gram}.
\]

\textbf{Vậy,} khối lượng kết tủa \ce{Cu(OH)2} là \SI{1.96}{\gram}.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Chuẩn độ \SI{25.0}{\milli\liter} dung dịch \ce{HCl} bằng dung dịch \ce{NaOH} \SI{0.100}{\mole\per\liter}. Ba lần chuẩn độ cần lần lượt 19,8 mL; 20,0 mL; 20,1 mL \ce{NaOH}. Lấy thể tích trung bình của ba lần, tính nồng độ mol của \ce{HCl}.
\dapso{$V_{tb}\approx\SI{19.97}{\milli\liter}$; $C_{\ce{HCl}}\approx\SI{0.0799}{\mole\per\liter}$.}
\begin{loigiaiBT}
\textbf{Bước 1. Tính số mol.}
\[
V_{\text{tb}}=\frac{19.8+20.0+20.1}{3}\approx\SI{19.97}{\milli\liter}.
\]
\[
n_{\ce{NaOH}}=0.01997\times0.100\approx\SI{1.997e-3}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c}
\ce{HCl} & + & \ce{NaOH} & \ce{->} & \ce{NaCl} & + & \ce{H2O}\\
$1{,}997\times10^{-3}$ & & $1{,}997\times10^{-3}$ & & $1{,}997\times10^{-3}$ & & $1{,}997\times10^{-3}$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\[
n_{\ce{HCl}}\approx\SI{1.997e-3}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
C_{\ce{HCl}}=\frac{1.997\times10^{-3}}{0.0250}\approx\SI{0.0799}{\mole\per\liter}.
\]

\textbf{Vậy,} nồng độ mol của dung dịch \ce{HCl} xấp xỉ \SI{0.0799}{\mole\per\liter}.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Một mẫu đá vôi chứa \SI{80}{\percent} \ce{CaCO3} theo khối lượng. Cho \SI{250}{\gram} mẫu phản ứng hoàn toàn với \ce{HCl} dư. Tính khối lượng \ce{CaCl2} tạo thành. Cho $M_{\ce{CaCO3}}=100$, $M_{\ce{CaCl2}}=111$ g/mol.
\dapso{\SI{222}{\gram} \ce{CaCl2}.}
\begin{loigiaiBT}
\textbf{Bước 1. Tính số mol.}
\[
m_{\ce{CaCO3}}=250\times0.80=\SI{200}{\gram},
\]
\[
n_{\ce{CaCO3}}=\frac{200}{100}=\SI{2.00}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c c c}
\ce{CaCO3} & + & \ce{2HCl} & \ce{->} & \ce{CaCl2} & + & \ce{CO2} & + & \ce{H2O}\\
$2{,}00$ & & & & $2{,}00$ & & $2{,}00$ & & $2{,}00$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\[
n_{\ce{CaCl2}}=\SI{2.00}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
m_{\ce{CaCl2}}=2.00\times111=\SI{222}{\gram}.
\]

\textbf{Vậy,} khối lượng \ce{CaCl2} tạo thành là \SI{222}{\gram}.
\end{loigiaiBT}
\end{tuluyen}
