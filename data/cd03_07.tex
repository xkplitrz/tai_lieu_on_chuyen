\section{Bài toán định lượng acid--base--oxide--muối}

\subsection{Các bước giải bài toán}

Các bài định lượng ở đây vẫn dùng một cách giải chung. Bài toán trở nên khó hơn chủ yếu vì bước tính toán trung gian dài hơn: phải xác định chất hết--dư, ghép nhiều phản ứng, đặt ẩn, lập hệ phương trình hoặc biện luận. Khi đã xử lí được các giá trị trung gian, bước cuối vẫn là tính đúng đại lượng đề hỏi và kết luận.

\begin{ghinho}{Sơ đồ giải}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c}
\fbox{\parbox{0.72\linewidth}{\centering \textbf{1. Tính số mol}}}\\[2mm]
$\downarrow$\\[2mm]
\fbox{\parbox{0.72\linewidth}{\centering \textbf{2. Viết phương trình hoặc sơ đồ phản ứng}}}\\[2mm]
$\downarrow$\\[2mm]
\fbox{\parbox{0.72\linewidth}{\centering \textbf{3. Tính các giá trị trung gian cần thiết}\\(số mol, khối lượng, chất hết--dư, ẩn số,\ldots)}}\\[2mm]
$\downarrow$\\[2mm]
\fbox{\parbox{0.72\linewidth}{\centering \textbf{4. Tính đại lượng đề hỏi và kết luận}}}
\end{tabular}
\end{center}
\end{ghinho}

\begin{notechem}
Không cần học nhiều ``dạng bài'' rời nhau. Hãy giữ nguyên bốn bước trên. Khi bài khó, phần cần suy nghĩ nhiều nhất thường nằm ở bước 3; đôi khi bước 4 cũng kéo dài nếu đề hỏi nồng độ sau phản ứng, độ tinh khiết, khối lượng dung dịch, lượng kết tủa, lượng khí hoặc phần trăm thành phần.
\end{notechem}

\subsection{Kĩ năng tính toán trực tiếp theo phương trình hóa học}

Khi số liệu đủ để xác định ngay lượng chất, ta chỉ cần đi lần lượt qua bốn bước. Nếu có nhiều phản ứng nối tiếp, kết quả của phản ứng trước trở thành dữ kiện cho phản ứng sau.

\begin{vidu}
Cho \SI{10.0}{\gram} \ce{CaCO3} tinh khiết phản ứng hoàn toàn với dung dịch \ce{HCl} dư. Tính thể tích \ce{CO2} ở điều kiện có $V_m=\SI{24.8}{\liter\per\mole}$.
\end{vidu}
\begin{loigiai}
\textbf{Bước 1. Tính số mol.}
\[
n_{\ce{CaCO3}}=\frac{10.0}{100}=\SI{0.100}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c c c}
\ce{CaCO3} & + & \ce{2HCl} & \ce{->} & \ce{CaCl2} & + & \ce{CO2} & + & \ce{H2O}\\
$0{,}100$ & & & & $0{,}100$ & & $0{,}100$ & & $0{,}100$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\[
n_{\ce{CO2}}=\SI{0.100}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
V_{\ce{CO2}}=0.100\times24.8=\SI{2.48}{\liter}.
\]

\textbf{Vậy,} thể tích \ce{CO2} thu được là \SI{2.48}{\liter}.
\end{loigiai}

\subsection{Kĩ năng xác định chất hết, chất dư và lượng sản phẩm}

Khi đề cho lượng của từ hai chất phản ứng trở lên, ở bước 3 phải xác định chất nào phản ứng hết trước. Lượng sản phẩm được tính theo chất hết; sau đó mới tính lượng chất còn dư nếu đề yêu cầu.

\begin{vidu}
Trộn \SI{200}{\milli\liter} dung dịch \ce{CuSO4} \SI{0.20}{\mole\per\liter} với \SI{100}{\milli\liter} dung dịch \ce{NaOH} \SI{0.50}{\mole\per\liter}. Tính khối lượng kết tủa. Cho $M_{\ce{Cu(OH)2}}=98$ g/mol.
\end{vidu}
\begin{loigiai}
\textbf{Bước 1. Tính số mol.}
\[
n_{\ce{CuSO4}}=0.200\times0.20=\SI{0.040}{\mole},
\]
\[
n_{\ce{NaOH}}=0.100\times0.50=\SI{0.050}{\mole}.
\]

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c}
\ce{CuSO4} & + & \ce{2NaOH} & \ce{->} & \ce{Cu(OH)2} & + & \ce{Na2SO4}\\
$0{,}040$ & & $0{,}050$ & & & & \\
$0{,}025$ & & $0{,}050$ & & $0{,}025$ & & $0{,}025$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}

\ce{NaOH} phản ứng hết trước, do đó:
\[
n_{\ce{Cu(OH)2}}=\frac{0.050}{2}=\SI{0.025}{\mole}.
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
m_{\ce{Cu(OH)2}}=0.025\times98=\SI{2.45}{\gram}.
\]

\textbf{Vậy,} khối lượng kết tủa \ce{Cu(OH)2} là \SI{2.45}{\gram}.
\end{loigiai}

\subsection{Kĩ năng lập hệ phương trình}

Khi hỗn hợp có nhiều chất chưa biết lượng, có thể đặt số mol mỗi chất làm ẩn. Mỗi dữ kiện độc lập tạo một phương trình. Nếu số phương trình độc lập bằng số ẩn, ta giải được hệ.

\begin{ghinho}{Cách thực hiện}
\begin{enumerate}
\item Ở bước 1, đặt số mol các chất chưa biết làm ẩn.
\item Ở bước 2, viết đầy đủ các phương trình hóa học và đặt số mol theo ẩn ngay dưới phương trình.
\item Ở bước 3, dùng các dữ kiện độc lập để lập hệ phương trình rồi giải hệ.
\item Ở bước 4, từ các ẩn đã tìm được tính đúng đại lượng đề hỏi và kết luận.
\end{enumerate}
\end{ghinho}

\begin{vidu}
Hòa tan hoàn toàn \SI{10.0}{\gram} hỗn hợp \ce{MgO} và \ce{CuO} bằng dung dịch \ce{HCl} vừa đủ. Sau phản ứng thu được \SI{19.625}{\gram} hỗn hợp muối chloride khan. Tính khối lượng mỗi oxide. Cho $M_{\ce{MgO}}=40$, $M_{\ce{MgCl2}}=95$, $M_{\ce{CuO}}=80$, $M_{\ce{CuCl2}}=135$ g/mol.
\end{vidu}
\begin{loigiai}
\textbf{Bước 1. Tính số mol.}

Gọi số mol \ce{MgO} và \ce{CuO} lần lượt là $x$ và $y$ mol.

\textbf{Bước 2. Viết phương trình hóa học và đặt số mol vào phương trình.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c c c}
\ce{MgO} & + & \ce{2HCl} & \ce{->} & \ce{MgCl2} & + & \ce{H2O}\\
$x$ & & $2x$ & & $x$ & & $x$\\[1mm]
\ce{CuO} & + & \ce{2HCl} & \ce{->} & \ce{CuCl2} & + & \ce{H2O}\\
$y$ & & $2y$ & & $y$ & & $y$
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian.}
\[
\begin{cases}
40x+80y=10.0,\\
95x+135y=19.625
\end{cases}
\Rightarrow
\begin{cases}
x=0.100,\\
y=0.075.
\end{cases}
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}
\[
m_{\ce{MgO}}=0.100\times40=\SI{4.0}{\gram},
\]
\[
m_{\ce{CuO}}=0.075\times80=\SI{6.0}{\gram}.
\]

\textbf{Vậy,} hỗn hợp ban đầu chứa \SI{4.0}{\gram} \ce{MgO} và \SI{6.0}{\gram} \ce{CuO}.
\end{loigiai}

Nếu có ba chất chưa biết và ba dữ kiện độc lập, nguyên tắc không thay đổi: đặt ba ẩn và lập ba phương trình. Phần này tập trung luyện chắc cách lập hệ; các bài có ba ẩn sẽ tiếp tục xuất hiện ở những chuyên đề hỗn hợp nhiều cấu tử.

\subsection{Kĩ năng biện luận từ điều kiện hóa học}

Có bài số ẩn nhiều hơn số phương trình độc lập. Khi đó chưa thể giải bằng đại số ngay mà phải dùng thêm điều kiện hóa học: hóa trị, công thức chất, tính tan, hiện tượng, giới hạn khối lượng mol hoặc tính nguyên của chỉ số trong công thức.

\begin{vidu}
Một acid C có khối lượng mol nhỏ hơn \SI{100}{\gram\per\mole}. Để trung hòa \SI{2.94}{\gram} C cần đúng \SI{0.060}{\mole} \ce{NaOH}. Biết C là một acid vô cơ quen thuộc. Xác định C.
\source{Phỏng theo cách biện luận ở Câu II, đề tuyển sinh chuyên Hóa KHTN Hà Nội 2026.}
\end{vidu}
\begin{loigiai}
\textbf{Bước 1. Tính số mol.}

Gọi $k$ là số mol \ce{NaOH} cần để trung hòa hoàn toàn $1$ mol acid C. Khi đó:
\[
n_C=\frac{0.060}{k}\ \text{mol}.
\]

\textbf{Bước 2. Viết sơ đồ phản ứng và đặt số mol vào sơ đồ.}
\begin{center}\small
\setlength{\tabcolsep}{3pt}
\begin{tabular}{c c c c c}
C & + & $k$\ce{NaOH} & $\longrightarrow$ & muối $+$ nước\\
$\dfrac{0{,}060}{k}$ & & $0{,}060$ & &
\end{tabular}
\end{center}

\textbf{Bước 3. Tính các giá trị trung gian và biện luận.}
\[
M_C=\frac{2.94}{0.060/k}=49k.
\]
Vì $M_C<100$ nên $k=1$ hoặc $k=2$.
\[
\begin{array}{c|c|c}
 k & M_C\,(\text{g/mol}) & \text{Kết luận}\\
\hline
1 & 49 & \text{không phù hợp}\\
2 & 98 & \ce{H2SO4}
\end{array}
\]

\textbf{Bước 4. Tính đại lượng đề hỏi và kết luận.}

\textbf{Vậy,} acid C là \ce{H2SO4}.
\end{loigiai}

\subsection{Kĩ năng chuyển số liệu thực nghiệm thành dữ kiện tính toán}

Ở các bài chuẩn độ, thay đổi khối lượng, lượng kết tủa hoặc lượng khí thoát ra, số liệu đo chưa phải là kết quả cuối cùng. Trước hết phải chuyển số liệu thực nghiệm thành số mol của một chất; sau đó tiếp tục theo đúng bốn bước giải chung.

\begin{ghinho}{Ví dụ với chuẩn độ}
\[
\begin{array}{c}
V_{\text{dung dịch chuẩn}}\\
\downarrow\\
n_{\text{dung dịch chuẩn}}\\
\downarrow\\
\text{PTHH và tỉ lệ mol}\\
\downarrow\\
n_{\text{chất cần tìm}}\\
\downarrow\\
C,\ m,\ \%\ldots
\end{array}
\]
\end{ghinho}
