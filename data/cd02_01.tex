\section{Phản ứng hóa học và năng lượng}

\subsection{Phản ứng hóa học và sự biến đổi năng lượng}

Trong một phản ứng hóa học, các chất đầu biến đổi thành các chất mới. Cùng với sự biến đổi về chất, phản ứng thường kèm theo sự trao đổi năng lượng với môi trường. Ở mức độ của chương trình THCS, năng lượng thường được xét dưới dạng nhiệt.

\begin{ghinho}{Phản ứng tỏa nhiệt và phản ứng thu nhiệt}
\begin{itemize}
\item \textbf{Phản ứng tỏa nhiệt} là phản ứng truyền năng lượng ra môi trường. Môi trường xung quanh thường nóng lên. Các phản ứng cháy của than, xăng, khí thiên nhiên, ethanol là những ví dụ quen thuộc.
\item \textbf{Phản ứng thu nhiệt} là phản ứng nhận năng lượng từ môi trường. Muốn phản ứng tiếp tục xảy ra thường phải cung cấp nhiệt từ bên ngoài, chẳng hạn quá trình nhiệt phân limestone:
\[
\ce{CaCO3 ->[t^\circ] CaO + CO2}.
\]
\end{itemize}
Không nên kết luận một phản ứng là thu nhiệt chỉ vì phải châm lửa hoặc đun nóng lúc đầu. Nhiều phản ứng tỏa nhiệt vẫn cần một lượng năng lượng ban đầu để khởi động.
\end{ghinho}

\begin{vidu}
Xét các quá trình sau:
\begin{enumerate}
\item đốt methane trong oxygen;
\item nung limestone để thu quicklime;
\item cho quicklime vào nước;
\item đốt ethanol làm nhiên liệu.
\end{enumerate}
Hãy phân loại các quá trình trên thành tỏa nhiệt hoặc thu nhiệt.
\end{vidu}
\begin{loigiai}
Các phản ứng cháy của methane và ethanol giải phóng nhiệt nên là phản ứng tỏa nhiệt. Phản ứng của quicklime với nước làm hệ nóng lên rõ rệt nên cũng là phản ứng tỏa nhiệt:
\[
\ce{CaO + H2O -> Ca(OH)2}.
\]
Quá trình nhiệt phân \ce{CaCO3} cần được cung cấp nhiệt liên tục:
\[
\ce{CaCO3 ->[t^\circ] CaO + CO2},
\]
nên là quá trình thu nhiệt.

\textbf{Vậy,} các quá trình (1), (3), (4) là tỏa nhiệt; quá trình (2) là thu nhiệt.
\end{loigiai}

\subsection{Lượng nhiệt của phản ứng}

Trong đề tuyển sinh, năng lượng thường được cung cấp dưới dạng ``mỗi mol chất cháy tỏa ra bao nhiêu kJ'', ``mỗi kilogram nhiên liệu cho bao nhiêu MJ'', hoặc một bảng số liệu tương tự. Khi đó, không cần học thêm một công thức nhiệt hóa học mới; chỉ cần hiểu đúng ý nghĩa của số liệu và dùng tỉ lệ.

Nếu \(q\) là nhiệt lượng tỏa ra khi một mol chất phản ứng hoàn toàn thì với \(n\) mol:
\[
Q=nq.
\]
Nếu dữ kiện cho nhiệt lượng trên một đơn vị khối lượng, chẳng hạn \(q_m\) kJ/kg, thì:
\[
Q=mq_m.
\]

\begin{ghinho}{Đọc đúng đơn vị năng lượng}
\[
\SI{1}{\mega\joule}=\SI{1000}{\kilo\joule},\qquad
\SI{1}{\kilo\joule}=\SI{1000}{\joule}.
\]
Trong một bài toán, nên đưa các giá trị năng lượng về cùng đơn vị trước khi so sánh hoặc cộng lại.
\end{ghinho}

\begin{vidu}
Khi đốt cháy hoàn toàn \(1\) mol methane, phản ứng tỏa ra \SI{890}{\kilo\joule}. Tính nhiệt lượng tỏa ra khi đốt cháy hoàn toàn \SI{3.20}{\kilo\gram} methane. Cho \(M_{\ce{CH4}}=16\) g/mol.
\end{vidu}
\begin{loigiai}
Đổi \SI{3.20}{\kilo\gram} \(=\SI{3200}{\gram}\). Số mol methane:
\[
n_{\ce{CH4}}=\frac{3200}{16}=\SI{200}{\mole}.
\]
Nhiệt lượng tỏa ra:
\[
Q=200\times890=\SI{178000}{\kilo\joule}=\SI{178}{\mega\joule}.
\]

\textbf{Vậy,} khi đốt cháy hoàn toàn \SI{3.20}{\kilo\gram} methane, nhiệt lượng tỏa ra là \SI{178}{\mega\joule}.
\end{loigiai}

\begin{vidu}
Một loại than có nhiệt trị \SI{28}{\mega\joule\per\kilo\gram}. Tính khối lượng than cần đốt để cung cấp \SI{420}{\mega\joule} nhiệt, giả sử toàn bộ nhiệt lượng được sử dụng.
\end{vidu}
\begin{loigiai}
Khối lượng than cần dùng:
\[
m=\frac{420}{28}=\SI{15}{\kilo\gram}.
\]

\textbf{Vậy,} cần đốt \SI{15}{\kilo\gram} than.
\end{loigiai}

\begin{tuluyen}
Khi đốt cháy hoàn toàn \(1\) mol propane, phản ứng tỏa ra \SI{2220}{\kilo\joule}. Tính nhiệt lượng tỏa ra khi đốt cháy \SI{8.8}{\kilo\gram} propane. Cho \(M_{\ce{C3H8}}=44\) g/mol.
\dapso{\SI{444}{\mega\joule}.}
\begin{loigiaiBT}
Số mol propane:
\[
n=\frac{8800}{44}=\SI{200}{\mole}.
\]
Nhiệt lượng tỏa ra:
\[
Q=200\times2220=\SI{444000}{\kilo\joule}=\SI{444}{\mega\joule}.
\]
\textbf{Vậy,} nhiệt lượng tỏa ra là \SI{444}{\mega\joule}.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Một loại củi khô có nhiệt trị \SI{15}{\mega\joule\per\kilo\gram}. Tính nhiệt lượng tỏa ra khi đốt cháy hoàn toàn \SI{24}{\kilo\gram} củi.
\dapso{\SI{360}{\mega\joule}.}
\begin{loigiaiBT}
\[
Q=24\times15=\SI{360}{\mega\joule}.
\]
\textbf{Vậy,} nhiệt lượng tỏa ra là \SI{360}{\mega\joule}.
\end{loigiaiBT}
\end{tuluyen}
