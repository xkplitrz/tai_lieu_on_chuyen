\section{Năng lượng trong phản ứng hóa học}

\subsection{Phản ứng tỏa nhiệt và phản ứng thu nhiệt}

Trong phản ứng hóa học, ngoài sự biến đổi từ chất đầu thành sản phẩm còn có sự trao đổi năng lượng với môi trường. Ở mức độ THCS, năng lượng thường được xét dưới dạng nhiệt.

\begin{ghinho}{Phản ứng tỏa nhiệt và phản ứng thu nhiệt}
\begin{itemize}
\item \textbf{Phản ứng tỏa nhiệt} là phản ứng truyền năng lượng ra môi trường. Các phản ứng cháy của than, xăng, khí thiên nhiên, ethanol là những ví dụ quen thuộc.
\item \textbf{Phản ứng thu nhiệt} là phản ứng nhận năng lượng từ môi trường. Muốn phản ứng tiếp tục xảy ra thường phải cung cấp nhiệt, chẳng hạn:
\[
\ce{CaCO3 ->[t^\circ] CaO + CO2}.
\]
\end{itemize}
Không nên kết luận một phản ứng là thu nhiệt chỉ vì phải châm lửa hoặc đun nóng lúc đầu. Nhiều phản ứng tỏa nhiệt vẫn cần một lượng năng lượng ban đầu để khởi động.
\end{ghinho}

\begin{vidu}
Phân loại các quá trình sau thành tỏa nhiệt hoặc thu nhiệt: (1) đốt methane; (2) nung đá vôi; (3) cho vôi sống vào nước; (4) đốt ethanol.
\end{vidu}
\begin{loigiai}
Các phản ứng cháy methane và ethanol giải phóng nhiệt nên là phản ứng tỏa nhiệt. Phản ứng của vôi sống với nước cũng làm hệ nóng lên:
\[
\ce{CaO + H2O -> Ca(OH)2}.
\]
Nhiệt phân calcium carbonate cần được cung cấp nhiệt:
\[
\ce{CaCO3 ->[t^\circ] CaO + CO2}.
\]
\textbf{Vậy,} các quá trình (1), (3), (4) là tỏa nhiệt; quá trình (2) là thu nhiệt.
\end{loigiai}

\subsection{Các bước giải bài toán năng lượng}

Các bài toán năng lượng có thể dài hoặc ngắn rất khác nhau, nhưng phần lớn đều là những đoạn khác nhau của cùng một con đường. Có bài chỉ đi qua một bước; có bài phải đi qua nhiều bước liên tiếp.

\begin{ghinho}{Sơ đồ giải chung}
\[
\boxed{\text{Lượng chất hoặc nhiên liệu}}
\longleftrightarrow
\boxed{\text{Năng lượng tỏa ra}}
\]
\[
\boxed{\text{Năng lượng tỏa ra}}
\longleftrightarrow
\boxed{\text{Năng lượng có ích}}
\longleftrightarrow
\boxed{\text{Mục tiêu thực tế}}
\]
Trong đó, ``mục tiêu thực tế'' có thể là nhiệt dùng để đun nóng, quãng đường xe đi được, số ngày sử dụng, chi phí, lượng \ce{CO2} phát thải hoặc sản lượng của một quá trình.
\end{ghinho}

Khi gặp một bài toán, học sinh nên thực hiện bốn thao tác theo thứ tự sau.

\begin{enumerate}
\item \textbf{Xác định điểm đầu và điểm đích.} Đề cho đại lượng nào? Đề hỏi đại lượng nào?
\item \textbf{Chuẩn hóa lượng chất.} Nếu cần, đổi thể tích sang khối lượng, khối lượng sang số mol hoặc tách các thành phần của hỗn hợp.
\item \textbf{Đi qua bước năng lượng.} Tính năng lượng tỏa ra, năng lượng cần thiết hoặc năng lượng có ích; xử lí hiệu suất nếu có.
\item \textbf{Đổi sang đại lượng cần hỏi.} Từ năng lượng suy ra khối lượng nhiên liệu, quãng đường, số ngày, chi phí, lượng khí thải,... rồi viết câu kết luận.
\end{enumerate}

\begin{notechem}
Không phải bài nào cũng cần đủ bốn bước. Nếu đề đã cho trực tiếp khối lượng nhiên liệu và nhiệt trị thì có thể bắt đầu ngay ở bước năng lượng. Nếu đề hỏi trực tiếp nhiệt lượng thì có thể dừng sau bước 3. Điều quan trọng là xác định đúng \textbf{điểm bắt đầu} và \textbf{điểm kết thúc} trên sơ đồ.
\end{notechem}

\subsection{Xử lí dữ kiện nhiệt lượng}

Trong đề tuyển sinh, năng lượng thường được cho dưới một trong các dạng sau:
\begin{itemize}
\item một mol chất khi cháy tỏa ra bao nhiêu \si{\kilo\joule};
\item một kilogram nhiên liệu tỏa ra bao nhiêu \si{\mega\joule};
\item một gam nhiên liệu tỏa ra bao nhiêu \si{\kilo\joule};
\item một quá trình cần bao nhiêu năng lượng cho mỗi mol hoặc mỗi đơn vị khối lượng chất.
\end{itemize}

Nếu một mol chất tỏa ra nhiệt lượng $q$ thì:
\[
Q=nq.
\]
Nếu nhiệt trị theo khối lượng là $q_m$ thì:
\[
Q=mq_m.
\]

\begin{ghinho}{Đơn vị năng lượng thường gặp}
\[
\SI{1}{\mega\joule}=\SI{1000}{\kilo\joule},\qquad
\SI{1}{\kilo\joule}=\SI{1000}{\joule}.
\]
Trước khi cộng, trừ hoặc so sánh các giá trị năng lượng, phải đưa chúng về cùng đơn vị.
\end{ghinho}

\begin{vidu}
Khi đốt cháy hoàn toàn $1$ mol methane, phản ứng tỏa ra \SI{890}{\kilo\joule}. Tính nhiệt lượng tỏa ra khi đốt cháy hoàn toàn \SI{3.20}{\kilo\gram} methane. Cho $M_{\ce{CH4}}=16$ g/mol.
\end{vidu}
\begin{loigiai}
\textbf{Bước 1.} Điểm đầu là khối lượng methane; điểm đích là nhiệt lượng tỏa ra.

\textbf{Bước 2.} Đổi khối lượng methane thành số mol:
\[
\SI{3.20}{\kilo\gram}=\SI{3200}{\gram},\qquad
n_{\ce{CH4}}=\frac{3200}{16}=\SI{200}{\mole}.
\]

\textbf{Bước 3.} Tính nhiệt lượng:
\[
Q=200\times890=\SI{178000}{\kilo\joule}=\SI{178}{\mega\joule}.
\]

\textbf{Vậy,} khi đốt cháy hoàn toàn \SI{3.20}{\kilo\gram} methane, nhiệt lượng tỏa ra là \SI{178}{\mega\joule}.
\end{loigiai}

\begin{vidu}
Một loại than có nhiệt trị \SI{28}{\mega\joule\per\kilo\gram}. Tính khối lượng than cần đốt để tỏa ra \SI{420}{\mega\joule} nhiệt.
\end{vidu}
\begin{loigiai}
Điểm đầu của bài này là năng lượng cần có; điểm đích là khối lượng nhiên liệu. Do đó đi ngược trên sơ đồ:
\[
m=\frac{Q}{q_m}=\frac{420}{28}=\SI{15}{\kilo\gram}.
\]
\textbf{Vậy,} cần đốt \SI{15}{\kilo\gram} than để tỏa ra \SI{420}{\mega\joule} nhiệt.
\end{loigiai}

\begin{tuluyen}
Khi đốt cháy hoàn toàn $1$ mol propane, phản ứng tỏa ra \SI{2220}{\kilo\joule}. Tính nhiệt lượng tỏa ra khi đốt cháy \SI{8.8}{\kilo\gram} propane. Cho $M_{\ce{C3H8}}=44$ g/mol.
\dapso{\SI{444}{\mega\joule}.}
\begin{loigiaiBT}
\[
n=\frac{8800}{44}=\SI{200}{\mole},\qquad
Q=200\times2220=\SI{444000}{\kilo\joule}=\SI{444}{\mega\joule}.
\]
\textbf{Vậy,} nhiệt lượng tỏa ra là \SI{444}{\mega\joule}.
\end{loigiaiBT}
\end{tuluyen}
