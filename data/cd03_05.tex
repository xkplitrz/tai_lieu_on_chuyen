\section{Muối và phản ứng trong dung dịch}

\subsection{Muối và tính tan}

Muối là hợp chất gồm cation kim loại hoặc \ce{NH4+} và anion gốc acid. Khi giải bài phản ứng trong dung dịch, cần biết chất nào tan để xác định các chất có thể phản ứng.

Không cần học thuộc toàn bộ bảng tính tan. Hãy tra bảng khi cần và ghi nhớ dần các chất xuất hiện thường xuyên.

\begin{ghinho}{Dấu hiệu giúp nhận ra phản ứng}
Hai chất trong dung dịch thường phản ứng khi tạo được ít nhất một trong các sản phẩm sau:
\begin{itemize}
\item chất kết tủa;
\item chất khí;
\item nước hoặc chất điện li yếu.
\end{itemize}
\end{ghinho}

\subsection{Các phản ứng thường gặp của muối}

\textbf{Muối + kim loại:}
\[
\ce{CuSO4 + Fe -> FeSO4 + Cu}.
\]

\textbf{Muối + acid:}
\[
\ce{Na2CO3 + 2HCl -> 2NaCl + CO2 ^ + H2O}.
\]

\textbf{Muối + base:}
\[
\ce{CuSO4 + 2NaOH -> Cu(OH)2 v + Na2SO4}.
\]

\textbf{Muối + muối:}
\[
\ce{BaCl2 + Na2SO4 -> BaSO4 v + 2NaCl}.
\]

\subsection{Dự đoán phản ứng và hiện tượng}

\begin{ghinho}{Các bước thực hiện}
\begin{enumerate}
\item Viết công thức các chất ban đầu.
\item Dự đoán sản phẩm trao đổi hoặc sản phẩm của phản ứng thế.
\item Dùng bảng tính tan và tính chất của sản phẩm để kiểm tra phản ứng có xảy ra không.
\item Từ sản phẩm suy ra hiện tượng: kết tủa, khí, chất rắn bám, đổi màu dung dịch, chất rắn tan dần,\ldots
\end{enumerate}
\end{ghinho}

\begin{vidu}
Trộn dung dịch \ce{BaCl2} với dung dịch \ce{Na2SO4}. Nêu hiện tượng và viết phương trình hóa học.
\end{vidu}
\begin{loigiai}
Hai muối trao đổi ion, tạo \ce{BaSO4} không tan:
\[
\ce{BaCl2 + Na2SO4 -> BaSO4 v + 2NaCl}.
\]
Xuất hiện kết tủa trắng.

\textbf{Vậy,} phản ứng tạo kết tủa trắng \ce{BaSO4}.
\end{loigiai}

\subsection{Carbonate và bicarbonate}

Carbonate và bicarbonate xuất hiện nhiều trong các bài về đá vôi, nước cứng, xử lí cặn và các hiện tượng trong tự nhiên.

\textbf{Carbonate tác dụng với acid:}
\[
\ce{Na2CO3 + 2HCl -> 2NaCl + CO2 ^ + H2O},
\]
\[
\ce{CaCO3 + 2HCl -> CaCl2 + CO2 ^ + H2O}.
\]

\textbf{Bicarbonate tác dụng với acid:}
\[
\ce{NaHCO3 + HCl -> NaCl + CO2 ^ + H2O},
\]
\[
\ce{Ca(HCO3)2 + 2HCl -> CaCl2 + 2CO2 ^ + 2H2O}.
\]

\textbf{Bicarbonate tác dụng với base:}
\[
\ce{NaHCO3 + NaOH -> Na2CO3 + H2O},
\]
\[
\ce{Ca(HCO3)2 + Ca(OH)2 -> 2CaCO3 v + 2H2O}.
\]

Hai nhóm phản ứng trên cho thấy ion \ce{HCO3-} có thể phản ứng với cả acid và base. Vì vậy, \ce{HCO3-} thể hiện \textbf{tính lưỡng tính acid--base}.

Có thể nhìn ngắn gọn theo hai hướng:
\[
\ce{HCO3- + H+ -> CO2 ^ + H2O},
\]
\[
\ce{HCO3- + OH- -> CO3^2- + H2O}.
\]

\textbf{Bicarbonate bị phân hủy khi đun nóng:}
\[
\ce{2NaHCO3 ->[t^\circ] Na2CO3 + CO2 ^ + H2O},
\]
\[
\ce{Ca(HCO3)2 ->[t^\circ] CaCO3 v + CO2 ^ + H2O},
\]
\[
\ce{Mg(HCO3)2 ->[t^\circ] MgCO3 v + CO2 ^ + H2O}.
\]

\subsection{Đá vôi, hang động và thạch nhũ}

Nước mưa và nước ngầm hòa tan một phần \ce{CO2}. Khi nước giàu \ce{CO2} đi qua đá vôi, \ce{CaCO3} ít tan chuyển thành calcium hydrogencarbonate tan:
\[
\ce{CaCO3 + CO2 + H2O -> Ca(HCO3)2}.
\]
Quá trình kéo dài làm đá vôi bị hòa tan, tạo khe nứt và hang động.

Khi nước chứa \ce{Ca(HCO3)2} nhỏ vào trong hang, một phần \ce{CO2} thoát ra khỏi dung dịch. Cân bằng chuyển theo chiều tạo lại \ce{CaCO3}:
\[
\ce{Ca(HCO3)2 -> CaCO3 v + CO2 ^ + H2O}.
\]
\ce{CaCO3} kết tủa chậm qua thời gian tạo thành thạch nhũ và măng đá.

Có thể biểu diễn hai quá trình bằng một cân bằng thuận nghịch:
\[
\ce{CaCO3 + CO2 + H2O <=> Ca(HCO3)2}.
\]

\begin{vidu}
Giải thích vì sao nước chứa \ce{CO2} có thể làm đá vôi bị hòa tan, nhưng khi nước đó nhỏ lâu ngày trong hang động lại có thể tạo thạch nhũ.
\end{vidu}
\begin{loigiai}
Nước chứa \ce{CO2} hòa tan đá vôi theo phản ứng:
\[
\ce{CaCO3 + CO2 + H2O -> Ca(HCO3)2}.
\]
Khi nước đi vào hang, \ce{CO2} thoát dần ra khỏi dung dịch; phản ứng xảy ra theo chiều ngược:
\[
\ce{Ca(HCO3)2 -> CaCO3 v + CO2 ^ + H2O}.
\]
\textbf{Vậy,} cùng một hệ carbonate--bicarbonate có thể gây hòa tan đá vôi hoặc tạo kết tủa \ce{CaCO3} tùy điều kiện.
\end{loigiai}

\begin{tuluyen}
Viết phương trình hóa học cho các quá trình sau:
\begin{tasks}(2)
\task \ce{NaHCO3 + HCl};
\task \ce{NaHCO3 + NaOH};
\task \ce{NaHCO3 ->[t^\circ]};
\task \ce{Ca(HCO3)2 + Ca(OH)2};
\task \ce{Ca(HCO3)2 ->[t^\circ]};
\task \ce{CaCO3 + CO2 + H2O}.
\end{tasks}
\dapso{Xem các phương trình trong phần lý thuyết.}
\end{tuluyen}
