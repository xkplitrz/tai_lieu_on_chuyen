\section{Nhiên liệu hỗn hợp và bài toán thực tiễn}

\subsection{Xác định thành phần hỗn hợp nhiên liệu}

Nhiên liệu thực tế thường là hỗn hợp. LPG có thể chứa propane và butane; xăng sinh học E5, E10 là hỗn hợp giữa ethanol và xăng truyền thống. Khi đó, bước khó nhất thường không phải công thức năng lượng mà là \textbf{tách đúng lượng từng thành phần}.

\begin{ghinho}{Hướng giải bài nhiên liệu hỗn hợp}
\begin{enumerate}
\item Xác định thành phần được cho theo \textbf{mol, thể tích hay khối lượng}.
\item Chọn một cơ sở tính thuận lợi: tổng số mol, tổng thể tích hoặc tổng khối lượng.
\item Tính lượng từng thành phần; nếu cần thì đổi $V\to m\to n$.
\item Tính năng lượng do từng thành phần cung cấp.
\item Cộng lại: $Q_{\text{tổng}}=Q_1+Q_2+\cdots$.
\item Tiếp tục đi trên sơ đồ chung đến đại lượng đề hỏi.
\end{enumerate}
\end{ghinho}

\begin{vidu}
Một hỗn hợp khí gồm $40\%$ propane và $60\%$ butane theo số mol. Khi đốt hoàn toàn, $1$ mol propane và $1$ mol butane lần lượt tỏa \SI{2220}{\kilo\joule} và \SI{2850}{\kilo\joule}. Tính nhiệt lượng tỏa ra khi đốt cháy $100$ mol hỗn hợp.
\end{vidu}
\begin{loigiai}
Trong $100$ mol hỗn hợp có $40$ mol propane và $60$ mol butane.
\[
Q=40\times2220+60\times2850=\SI{259800}{\kilo\joule}
=\SI{259.8}{\mega\joule}.
\]
\textbf{Vậy,} $100$ mol hỗn hợp LPG tỏa ra \SI{259.8}{\mega\joule}.
\end{loigiai}

\subsection{Tính toán liên quan đến năng lượng}

Sau khi tính được năng lượng có ích, nhiều bài toán chỉ còn một bước chia cho ``mức tiêu hao cho một đơn vị mục tiêu''.

Nếu một xe cần $q_s$ năng lượng có ích cho mỗi kilomet:
\[
s=\frac{Q_{\text{ích}}}{q_s}.
\]
Nếu một gia đình cần $q_d$ năng lượng có ích mỗi ngày:
\[
N=\frac{Q_{\text{ích}}}{q_d}.
\]
Nếu biết khối lượng nhiên liệu tiêu thụ và đơn giá, chi phí được tính trực tiếp từ lượng nhiên liệu đã dùng.

\begin{ghinho}{Hướng giải bài thực tế}
Hãy viết chuỗi tính trước khi bấm máy. Ví dụ:
\[
V_{\text{nhiên liệu}}
\to m
\to n
\to Q_{\text{tỏa}}
\to Q_{\text{ích}}
\to s.
\]
Nếu bài hỏi chi phí hoặc số ngày, chỉ thay ô cuối bằng đại lượng tương ứng. Việc viết chuỗi giúp tránh bỏ sót hiệu suất hoặc nhầm giữa năng lượng tỏa ra và năng lượng có ích.
\end{ghinho}

\begin{vidu}
Một xe dùng \SI{4.0}{\liter} nhiên liệu có $D=\SI{0.75}{\kilo\gram\per\liter}$ và nhiệt trị \SI{44}{\mega\joule\per\kilo\gram}. Hiệu suất động cơ là $30\%$. Xe cần \SI{600}{\kilo\joule} năng lượng cơ học để đi \SI{1}{\kilo\metre}. Tính quãng đường tối đa.
\end{vidu}
\begin{loigiai}
Khối lượng nhiên liệu:
\[
m=4.0\times0.75=\SI{3.0}{\kilo\gram}.
\]
Nhiệt lượng tỏa ra:
\[
Q_{\text{tỏa}}=3.0\times44=\SI{132}{\mega\joule}=\SI{132000}{\kilo\joule}.
\]
Năng lượng có ích:
\[
Q_{\text{ích}}=132000\times0.30=\SI{39600}{\kilo\joule}.
\]
Quãng đường:
\[
s=\frac{39600}{600}=\SI{66}{\kilo\metre}.
\]
\textbf{Vậy,} xe đi được tối đa khoảng \SI{66}{\kilo\metre}.
\end{loigiai}

\subsection{So sánh nhiên liệu trên cùng một cơ sở}

Các câu hỏi so sánh thường yêu cầu đánh giá chi phí, lượng \ce{CO2} phát thải hoặc lượng nhiên liệu tiêu thụ. Không được so sánh trực tiếp hai số liệu nếu chúng không cùng cơ sở.

\begin{ghinho}{Quy tắc chuẩn hóa trước khi so sánh}
\begin{enumerate}
\item Khoanh rõ điều kiện ``cùng'': cùng năng lượng có ích, cùng quãng đường, cùng thể tích hay cùng khối lượng.
\item Tính lượng mỗi nhiên liệu cần để đạt cùng mục tiêu đó.
\item Từ lượng nhiên liệu đã chuẩn hóa, tính chi phí hoặc lượng \ce{CO2}.
\item Cuối cùng mới kết luận nhiên liệu nào tốt hơn theo tiêu chí đề bài.
\end{enumerate}
\end{ghinho}

\begin{vidu}
Methane và propane cháy hoàn toàn:
\[
\ce{CH4 + 2O2 -> CO2 + 2H2O},
\]
\[
\ce{C3H8 + 5O2 -> 3CO2 + 4H2O}.
\]
Biết $1$ mol methane tỏa \SI{890}{\kilo\joule}; $1$ mol propane tỏa \SI{2220}{\kilo\joule}. So sánh số mol \ce{CO2} tạo ra khi mỗi nhiên liệu cung cấp \SI{4440}{\kilo\joule}, bỏ qua thất thoát.
\end{vidu}
\begin{loigiai}
Với methane:
\[
n_{\ce{CH4}}=\frac{4440}{890}\approx\SI{4.989}{\mole}.
\]
Theo phương trình, $1$ mol methane tạo $1$ mol \ce{CO2}, nên:
\[
n_{\ce{CO2}}^{(\ce{CH4})}\approx\SI{4.989}{\mole}.
\]

Với propane:
\[
n_{\ce{C3H8}}=\frac{4440}{2220}=\SI{2.00}{\mole}.
\]
Theo phương trình, $1$ mol propane tạo $3$ mol \ce{CO2}:
\[
n_{\ce{CO2}}^{(\ce{C3H8})}=\SI{6.00}{\mole}.
\]

\textbf{Vậy,} khi cung cấp cùng \SI{4440}{\kilo\joule}, methane tạo ít \ce{CO2} hơn propane theo các dữ kiện đã cho.
\end{loigiai}

\subsection{Tính năng lượng trong quy trình sản xuất}

Dạng khó nhất thường ghép nhiều đoạn của sơ đồ chung với phương trình hóa học. Một chất được sản xuất hoặc phân hủy, quá trình đó cần năng lượng, rồi năng lượng lại được cung cấp bởi một nhiên liệu khác.

\begin{ghinho}{Hướng giải bài chuỗi công nghệ}
\begin{enumerate}
\item Bắt đầu từ đại lượng mục tiêu của quá trình chính và dùng phương trình hóa học để tìm lượng chất cần xử lí.
\item Từ lượng chất đó, tính năng lượng quá trình cần nhận hoặc giải phóng.
\item Xử lí hiệu suất truyền nhiệt hoặc hiệu suất sử dụng năng lượng.
\item Tính ngược lượng nhiên liệu phải dùng.
\item Nếu đề còn hỏi phát thải, chi phí hoặc sản phẩm phụ, tiếp tục từ lượng nhiên liệu vừa tìm được.
\end{enumerate}
\end{ghinho}

\begin{notechem}
Ở các bài dài, không nên cố viết một biểu thức duy nhất ngay từ đầu. Học sinh THCS nên ghi tên từng đại lượng trung gian và đi lần lượt qua các ô của sơ đồ. Cách này dài hơn một chút nhưng giảm mạnh nguy cơ nhầm dữ kiện và nhầm hiệu suất.
\end{notechem}
