\section{Chuỗi phản ứng và nhận biết chất}

\subsection{Từ phản ứng riêng lẻ đến chuỗi phản ứng}

Một chuỗi phản ứng gồm nhiều phản ứng riêng lẻ nối tiếp nhau. Vì vậy, trước khi giải chuỗi dài cần thành thạo việc dự đoán sản phẩm và viết đúng từng phương trình.

\begin{ghinho}{Làm từng mũi tên}
Với mỗi bước $A\to B$:
\begin{enumerate}
\item xác định loại chất của $A$ và $B$;
\item chọn chất phản ứng hoặc điều kiện phù hợp;
\item viết và cân bằng phương trình;
\item kiểm tra sản phẩm có đúng chất cần tạo hay không.
\end{enumerate}
\end{ghinho}

\subsection{Luyện tập viết phương trình hóa học}

\begin{tuluyen}
Viết phương trình hóa học của các phản ứng sau nếu có:
\begin{tasks}(2)
	\task \ce{Mg + HCl};
	\task \ce{CuO + H2SO4};
	\task \ce{CaO + H2O};
	\task \ce{CO2 + Ca(OH)2} với lượng \ce{CO2} vừa đủ;
	\task \ce{SO2 + NaOH} tạo muối trung hòa;
	\task \ce{HCl + NaOH};
	\task \ce{H2SO4 + KOH};
	\task \ce{CuSO4 + NaOH};
	\task \ce{FeCl3 + NaOH};
	\task \ce{BaCl2 + Na2SO4}.
\end{tasks}
\dapso{Viết các phương trình cân bằng tương ứng; trường hợp có kết tủa hoặc khí cần ghi rõ trạng thái.}
\end{tuluyen}

\begin{tuluyen}
Viết phương trình hóa học của các phản ứng sau nếu có:
\begin{tasks}(2)
	\task \ce{Na2CO3 + HCl};
	\task \ce{CaCO3 + HCl};
	\task \ce{NaHCO3 + HCl};
	\task \ce{NaHCO3 + NaOH};
	\task \ce{Ca(HCO3)2 + Ca(OH)2};
	\task nhiệt phân \ce{NaHCO3};
	\task nhiệt phân \ce{Ca(HCO3)2};
	\task \ce{CaCO3 + CO2 + H2O};
	\task \ce{AgNO3 + NaCl};
	\task \ce{Ba(NO3)2 + Na2SO4}.
\end{tasks}
\dapso{Xem lại tính chất carbonate, bicarbonate và điều kiện phản ứng trong dung dịch.}
\end{tuluyen}

\begin{tuluyen}
Dự đoán hiện tượng và viết phương trình hóa học:
\begin{tasks}
	\task nhỏ \ce{NaOH} vào dung dịch \ce{CuSO4};
	\task nhỏ \ce{HCl} vào bột \ce{CaCO3};
	\task dẫn \ce{CO2} vừa đủ vào nước vôi trong;
	\task cho iron vào dung dịch \ce{CuSO4};
	\task trộn dung dịch \ce{BaCl2} và \ce{Na2SO4};
	\task đun dung dịch chứa \ce{Ca(HCO3)2}.
\end{tasks}
\dapso{Lần lượt xuất hiện kết tủa, khí, kết tủa, chất rắn bám/đổi màu dung dịch, kết tủa và cặn carbonate.}
\end{tuluyen}

\subsection{Chuỗi phản ứng}

\begin{vidu}
Hoàn thành chuỗi phản ứng:
\[
\begin{aligned}
\ce{CaCO3} &\ce{-> CaO -> Ca(OH)2 -> CaCO3}\\
&\ce{-> Ca(HCO3)2 -> CaCO3}.
\end{aligned}
\]
\end{vidu}
\begin{loigiai}
\[
\ce{CaCO3 ->[t^\circ] CaO + CO2},
\]
\[
\ce{CaO + H2O -> Ca(OH)2},
\]
\[
\ce{Ca(OH)2 + CO2 -> CaCO3 v + H2O},
\]
\[
\ce{CaCO3 + CO2 + H2O -> Ca(HCO3)2},
\]
\[
\ce{Ca(HCO3)2 -> CaCO3 v + CO2 ^ + H2O}.
\]
\textbf{Vậy,} mỗi mũi tên trong chuỗi là một phản ứng đã học riêng lẻ.
\end{loigiai}

\begin{tuluyen}
Hoàn thành các chuỗi phản ứng sau, ghi rõ chất phản ứng hoặc điều kiện cần dùng:
\begin{enumerate}
\item \ce{Cu -> CuO -> CuCl2 -> Cu(OH)2 -> CuO -> CuSO4};
\item \ce{MgCO3 -> MgO -> MgCl2 -> Mg(OH)2 -> MgO};
\item \ce{CaCO3 -> CaO -> Ca(OH)2 -> CaCO3 -> Ca(HCO3)2};
\item \ce{FeCl3 -> Fe(OH)3 -> Fe2O3 -> FeCl3};
\item \ce{CO2 -> CaCO3 -> Ca(HCO3)2 -> CaCO3};
\item \ce{NaHCO3 -> Na2CO3 -> CO2 -> CaCO3}.
\end{enumerate}
\dapso{Mỗi chuỗi có nhiều cách thực hiện; cần viết đủ phương trình và điều kiện cho từng mũi tên.}
\end{tuluyen}

\begin{tuluyen}
Hoàn thành chuỗi phản ứng sau:
\[
\ce{Al -> Al2O3 -> NaAlO2 -> Al(OH)3 -> Al2(SO4)3 -> BaSO4}.
\]
\source{Chuỗi mở rộng theo kiểu Câu II, đề tuyển sinh chuyên Hóa KHTN Hà Nội 2026.}
\dapso{Đây là chuỗi mở rộng; cần dùng lần lượt oxygen, \ce{NaOH}, \ce{CO2 + H2O}, \ce{H2SO4} và muối barium tan.}
\end{tuluyen}

\subsection{Nhận biết chất bằng hiện tượng}

Một bài nhận biết không bắt đầu bằng việc học thuộc thuốc thử. Hãy tìm tính chất khác nhau có thể quan sát được.

\begin{ghinho}{Hướng giải}
\begin{enumerate}
\item Liệt kê các chất cần phân biệt.
\item Chọn thuốc thử tạo các hiện tượng khác nhau.
\item Nếu chưa phân biệt hết, tiếp tục với các mẫu còn lại.
\item Ghi rõ hiện tượng và phương trình hóa học ở mỗi bước.
\end{enumerate}
\end{ghinho}

\begin{vidu}
Có ba dung dịch không nhãn: \ce{HCl}, \ce{NaOH}, \ce{NaCl}. Chỉ dùng quỳ tím, hãy nhận biết từng dung dịch.
\end{vidu}
\begin{loigiai}
Nhúng quỳ tím vào từng mẫu:
\begin{itemize}
\item mẫu làm quỳ tím hóa đỏ là \ce{HCl};
\item mẫu làm quỳ tím hóa xanh là \ce{NaOH};
\item mẫu không làm đổi màu quỳ tím là \ce{NaCl}.
\end{itemize}
\textbf{Vậy,} có thể nhận biết cả ba dung dịch chỉ bằng quỳ tím.
\end{loigiai}
