\section{Bài tập chuyên đề}

Hệ thống bài tập dưới đây được sắp xếp từ nhận biết hiện tượng đến xử lí dữ liệu và bài toán tích hợp. Các bài cuối có sử dụng kiểu dữ kiện thường gặp trong đề tuyển sinh chuyên của nhiều tỉnh, nhưng kiến thức cốt lõi vẫn xoay quanh phản ứng hóa học và tốc độ phản ứng.

\subsection{Củng cố}

\begin{tuluyen}
Trong các hiện tượng sau, hiện tượng nào là biến đổi hóa học: (a) iodine thăng hoa; (b) magnesium cháy trong oxygen; (c) nước đóng băng; (d) sữa để lâu bị chua; (e) nghiền nhỏ than?
\dapso{(b), (d).}
\begin{loigiaiBT}
Magnesium cháy tạo magnesium oxide; sữa bị chua do hình thành các chất mới. Các hiện tượng còn lại không tạo chất mới.

\textbf{Vậy,} (b) và (d) là biến đổi hóa học.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Cho phản ứng \(\ce{2Mg + O2 -> 2MgO}\). Nếu \SI{4.8}{\gram} Mg phản ứng vừa đủ với \SI{3.2}{\gram} \ce{O2}, tính khối lượng \ce{MgO} tạo thành bằng định luật bảo toàn khối lượng.
\dapso{\SI{8.0}{\gram}.}
\begin{loigiaiBT}
Theo định luật bảo toàn khối lượng:
\[
m_{\ce{MgO}}=4.8+3.2=\SI{8.0}{\gram}.
\]
\textbf{Vậy,} khối lượng \ce{MgO} tạo thành là \SI{8.0}{\gram}.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Nêu dấu hiệu quan sát được và viết phương trình hóa học khi cho Zn vào dung dịch \ce{HCl}.
\dapso{Zn tan dần, có bọt khí \ce{H2}; \(\ce{Zn + 2HCl -> ZnCl2 + H2}\).}
\begin{loigiaiBT}
Zn phản ứng với hydrochloric acid tạo muối zinc chloride và hydrogen. Quan sát thấy kim loại tan dần và có bọt khí thoát ra.

\textbf{Vậy,} dấu hiệu phản ứng là Zn tan và có khí \ce{H2}; phương trình là \(\ce{Zn + 2HCl -> ZnCl2 + H2}\).
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Trong bốn biện pháp sau, biện pháp nào chủ yếu làm tăng tốc độ phản ứng nhờ tăng diện tích bề mặt: (1) chẻ nhỏ củi; (2) tăng nhiệt độ lò; (3) nghiền quặng trước khi xử lí; (4) dùng chất xúc tác?
\dapso{(1) và (3).}
\begin{loigiaiBT}
Chẻ nhỏ củi và nghiền quặng làm tăng tổng diện tích bề mặt chất rắn tiếp xúc với chất phản ứng khác.

\textbf{Vậy,} các biện pháp (1) và (3) làm tăng tốc độ chủ yếu do tăng diện tích bề mặt.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Khi cho cùng khối lượng Zn vào cùng thể tích dung dịch \ce{HCl} dư, thí nghiệm A dùng acid \SI{0.5}{\mole\per\liter}, thí nghiệm B dùng acid \SI{1.5}{\mole\per\liter}. Các điều kiện khác như nhau. Thí nghiệm nào kết thúc sớm hơn? Lượng \ce{H2} cuối cùng của hai thí nghiệm có khác nhau không?
\dapso{B kết thúc sớm hơn; nếu \ce{HCl} đều dư và Zn như nhau thì lượng \ce{H2} cuối cùng bằng nhau.}
\begin{loigiaiBT}
Nồng độ acid cao hơn làm phản ứng xảy ra nhanh hơn nên B kết thúc sớm hơn. Tuy nhiên, do Zn có cùng lượng và acid đều dư, Zn là chất quyết định lượng \ce{H2} cuối cùng.

\textbf{Vậy,} thí nghiệm B nhanh hơn nhưng hai thí nghiệm tạo cùng lượng \ce{H2} khi phản ứng hoàn toàn.
\end{loigiaiBT}
\end{tuluyen}

\subsection{Vận dụng}

\begin{tuluyen}
Phản ứng tạo khí được theo dõi bằng thể tích khí:
\begin{center}
\begin{tabular}{c|ccccc}
$t$ (s) & 0 & 15 & 30 & 45 & 60 \\
\hline
$V$ (mL) & 0 & 33 & 57 & 72 & 78
\end{tabular}
\end{center}
Tính tốc độ tăng thể tích khí trung bình trong các khoảng 0--15 s và 45--60 s; từ đó nhận xét.
\source{Phỏng theo dạng bảng số liệu tốc độ phản ứng trong đề chuyên Hóa Điện Biên 2026.}
\dapso{\SI{2.2}{\milli\liter\per\second}; \SI{0.4}{\milli\liter\per\second}; phản ứng chậm dần.}
\begin{loigiaiBT}
\[
\overline v_{0-15}=\frac{33}{15}=\SI{2.2}{\milli\liter\per\second},
\]
\[
\overline v_{45-60}=\frac{78-72}{15}=\SI{0.4}{\milli\liter\per\second}.
\]
Tốc độ trung bình ở giai đoạn sau nhỏ hơn.

\textbf{Vậy,} phản ứng chậm dần theo thời gian.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Nồng độ chất A trong một phản ứng giảm từ \SI{0.90}{\mole\per\liter} xuống \SI{0.54}{\mole\per\liter} trong 20 s đầu, sau đó giảm từ \SI{0.54}{\mole\per\liter} xuống \SI{0.42}{\mole\per\liter} trong 20 s tiếp theo. Tính độ giảm nồng độ trung bình trong mỗi giai đoạn và so sánh tốc độ.
\source{Phỏng theo dạng đồ thị nồng độ--thời gian trong đề chuyên Hóa Đà Nẵng 2026.}
\dapso{\SI{0.018}{\mole\per\liter\per\second}; \SI{0.006}{\mole\per\liter\per\second}; giai đoạn đầu nhanh gấp 3 lần.}
\begin{loigiaiBT}
\[
\overline v_1=\frac{0.90-0.54}{20}=\SI{0.018}{\mole\per\liter\per\second},
\]
\[
\overline v_2=\frac{0.54-0.42}{20}=\SI{0.006}{\mole\per\liter\per\second}.
\]
Tỉ số \(\overline v_1/\overline v_2=3\).

\textbf{Vậy,} giai đoạn đầu phản ứng diễn ra nhanh gấp 3 lần giai đoạn sau theo đại lượng đang theo dõi.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Một học sinh khảo sát phản ứng Zn với \ce{HCl} bằng bốn thí nghiệm. Hãy chọn cặp phù hợp nhất để khảo sát riêng ảnh hưởng của nhiệt độ:
\begin{center}
\begin{tabular}{c|ccc}
Thí nghiệm & Zn & $C_M$(\ce{HCl}) & Nhiệt độ \\
\hline
A & \SI{2.0}{\gram}, viên & \SI{1.0}{\mole\per\liter} & \SI{20}{\degreeCelsius} \\
B & \SI{2.0}{\gram}, viên & \SI{1.0}{\mole\per\liter} & \SI{40}{\degreeCelsius} \\
C & \SI{2.0}{\gram}, bột & \SI{1.0}{\mole\per\liter} & \SI{40}{\degreeCelsius} \\
D & \SI{2.0}{\gram}, viên & \SI{2.0}{\mole\per\liter} & \SI{40}{\degreeCelsius} \\
\end{tabular}
\end{center}
\source{Phỏng theo dạng thiết kế thí nghiệm trong đề chuyên Hóa Đắk Lắk 2026.}
\dapso{A và B.}
\begin{loigiaiBT}
A và B có cùng khối lượng và dạng Zn, cùng nồng độ acid; chỉ nhiệt độ thay đổi.

\textbf{Vậy,} cặp A--B phù hợp để khảo sát riêng ảnh hưởng của nhiệt độ.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Trong phản ứng phân hủy \ce{H2O2}, khi thêm một lượng nhỏ \ce{MnO2}, khí \ce{O2} thoát ra nhanh hơn nhưng sau phản ứng vẫn có thể thu lại \ce{MnO2}. Hãy nêu vai trò của \ce{MnO2} và giải thích vì sao không được tính \ce{MnO2} là chất phản ứng tiêu hao theo phương trình tổng quát.
\dapso{\ce{MnO2} là chất xúc tác; nó làm tăng tốc độ phản ứng nhưng không bị tiêu hao theo phương trình tổng quát.}
\begin{loigiaiBT}
\ce{MnO2} làm phản ứng phân hủy hydrogen peroxide diễn ra nhanh hơn và được tái tạo sau quá trình xúc tác nên không bị tiêu hao tổng thể.

\textbf{Vậy,} \ce{MnO2} đóng vai trò chất xúc tác, không phải chất phản ứng tiêu hao theo phương trình tổng quát.
\end{loigiaiBT}
\end{tuluyen}

\subsection{Tổng hợp và mở rộng}

\begin{tuluyen}
Một phản ứng có hệ số nhiệt độ \(\gamma=2\). Ở \SI{20}{\degreeCelsius}, cần 24 phút để đạt một mức chuyển hóa xác định. Nếu chỉ thay đổi nhiệt độ lên \SI{50}{\degreeCelsius}, ước tính thời gian cần thiết.
\source{Phỏng theo dạng quy tắc Van't Hoff trong đề tuyển sinh chuyên Hóa Hà Nội 2025.}
\dapso{3 phút.}
\begin{loigiaiBT}
Tăng \SI{30}{\degreeCelsius} tương ứng ba khoảng \SI{10}{\degreeCelsius}, nên tốc độ tăng \(2^3=8\) lần. Thời gian để đạt cùng mức chuyển hóa giảm 8 lần:
\[
t=\frac{24}{8}=3\ \text{phút}.
\]
\textbf{Vậy,} thời gian ước tính là 3 phút.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Hai thí nghiệm đều dùng \SI{5.0}{\gram} \ce{CaCO3} và lượng \ce{HCl} dư. Thí nghiệm I dùng đá vôi dạng cục với \ce{HCl} \SI{1.0}{\mole\per\liter} ở \SI{25}{\degreeCelsius}; thí nghiệm II dùng đá vôi dạng bột với \ce{HCl} \SI{2.0}{\mole\per\liter} ở \SI{40}{\degreeCelsius}. Một học sinh kết luận: ``Do nghiền nhỏ đá vôi nên thí nghiệm II nhanh hơn''. Hãy đánh giá kết luận.
\dapso{Kết luận chưa đủ căn cứ vì đồng thời thay đổi diện tích bề mặt, nồng độ và nhiệt độ.}
\begin{loigiaiBT}
Giữa hai thí nghiệm có ba yếu tố thay đổi: mức độ chia nhỏ \ce{CaCO3}, nồng độ acid và nhiệt độ. Cả ba đều có thể làm thay đổi tốc độ phản ứng.

\textbf{Vậy,} không thể quy riêng sự tăng tốc độ cho việc nghiền nhỏ đá vôi; cần thiết kế thí nghiệm chỉ thay đổi một yếu tố.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Trong phản ứng tạo khí, sau 30 s thu được \SI{72}{\milli\liter} khí ở thí nghiệm A và \SI{96}{\milli\liter} khí ở thí nghiệm B. Sau thời gian đủ dài, cả hai thí nghiệm đều thu được \SI{120}{\milli\liter} khí. Biết hai thí nghiệm dùng cùng lượng chất phản ứng và chỉ khác một điều kiện. Có thể kết luận gì về tốc độ và lượng sản phẩm cuối cùng?
\dapso{B nhanh hơn trong 30 s đầu; lượng sản phẩm cuối cùng bằng nhau.}
\begin{loigiaiBT}
Trong cùng 30 s, B tạo nhiều khí hơn nên B có tốc độ trung bình lớn hơn. Tuy nhiên cả hai cùng đạt \SI{120}{\milli\liter} khi phản ứng hoàn tất.

\textbf{Vậy,} điều kiện ở B làm phản ứng diễn ra nhanh hơn nhưng không làm thay đổi lượng khí cuối cùng trong trường hợp này.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Một phản ứng được theo dõi bằng nồng độ chất A:
\begin{center}
\begin{tabular}{c|ccccccc}
$t$ (s) & 0 & 10 & 20 & 30 & 40 & 50 & 60 \\
\hline
$C_A$ (mol/L) & 1.00 & 0.74 & 0.56 & 0.44 & 0.36 & 0.31 & 0.28
\end{tabular}
\end{center}
\begin{enumerate}
\item Tính độ giảm nồng độ trung bình của A trong mỗi khoảng 10 s.
\item Khoảng nào phản ứng nhanh nhất? Khoảng nào chậm nhất?
\item Dự đoán hình dạng định tính của đồ thị $C_A$ theo thời gian.
\end{enumerate}
\source{Phỏng theo cách khai thác đồ thị tốc độ trong đề chuyên Hóa Đà Nẵng 2026 và Đắk Lắk 2026.}
\dapso{0.026; 0.018; 0.012; 0.008; 0.005; 0.003 mol/L/s; nhanh nhất 0--10 s, chậm nhất 50--60 s; đồ thị giảm nhanh lúc đầu rồi thoải dần.}
\begin{loigiaiBT}
Các độ giảm nồng độ trung bình lần lượt là:
\[
0.026;\ 0.018;\ 0.012;\ 0.008;\ 0.005;\ 0.003\ \si{\mole\per\liter\per\second}.
\]
Giá trị lớn nhất ở 0--10 s, nhỏ nhất ở 50--60 s. Vì mức giảm theo thời gian nhỏ dần, đồ thị đi xuống nhưng độ dốc giảm dần về độ lớn.

\textbf{Vậy,} phản ứng nhanh nhất lúc đầu, chậm dần theo thời gian; đồ thị $C_A(t)$ giảm mạnh lúc đầu rồi thoải dần.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Trong sản xuất, một phản ứng được tiến hành ở hai nhiệt độ. Ở \SI{30}{\degreeCelsius}, sau 10 phút đạt 25\% mức chuyển hóa mong muốn; ở \SI{50}{\degreeCelsius}, sau 10 phút đạt 70\%. Một học sinh phát biểu: ``Tăng nhiệt độ làm lượng sản phẩm cuối cùng chắc chắn tăng 2,8 lần''. Hãy nhận xét.
\dapso{Sai; dữ kiện chỉ cho thấy phản ứng diễn ra nhanh hơn trong 10 phút, không đủ để kết luận sản phẩm cuối cùng tăng 2,8 lần.}
\begin{loigiaiBT}
Sau cùng một thời gian, hệ ở nhiệt độ cao đạt mức chuyển hóa lớn hơn nên có thể kết luận tốc độ phản ứng lớn hơn. Tuy nhiên dữ kiện không cho biết trạng thái cuối cùng khi phản ứng hoàn tất.

\textbf{Vậy,} phát biểu sai; chỉ có thể kết luận tăng nhiệt độ làm phản ứng nhanh hơn trong điều kiện đã cho.
\end{loigiaiBT}
\end{tuluyen}

\begin{tuluyen}
Một viên sủi được thả vào \SI{200}{\milli\liter} nước ở \SI{20}{\degreeCelsius}; viên tan hết sau 180 s. Hãy đề xuất hai cách làm viên sủi phản ứng nhanh hơn mà không thay đổi lượng viên sủi, đồng thời giải thích bằng kiến thức tốc độ phản ứng. Nếu muốn khảo sát riêng từng cách, cần kiểm soát những điều kiện nào?
\dapso{Ví dụ: dùng nước ấm hơn; nghiền nhỏ viên sủi. Khi khảo sát từng yếu tố, giữ các điều kiện còn lại như lượng nước, nhiệt độ hoặc kích thước tương ứng không đổi.}
\begin{loigiaiBT}
Có thể tăng nhiệt độ nước hoặc làm nhỏ viên sủi để tăng diện tích bề mặt. Khi khảo sát nhiệt độ, phải giữ nguyên lượng nước và kích thước viên; khi khảo sát diện tích bề mặt, phải giữ nguyên lượng nước và nhiệt độ.

\textbf{Vậy,} có thể tăng nhiệt độ hoặc tăng diện tích bề mặt để làm phản ứng nhanh hơn, nhưng mỗi thí nghiệm so sánh chỉ nên thay đổi một yếu tố.
\end{loigiaiBT}
\end{tuluyen}
